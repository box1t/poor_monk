```c++
string reverse(string const &s) {
    string rev; 
    for (int i = s.size() - 1; i >= 0; i--) {
        rev = rev.append(1, s[i]); // добавляем (append - добавить) элемент s[i] единожды
    }
    return rev;
}
```
