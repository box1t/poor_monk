# 9. (Extra) Высота двоичного дерева

```c++
struct Node {
	int data;
	Node* left;
	Node* right;
};

int max(x, y) {
	if (x >= y) {
		return x;	
	} else {
		return y;
}

int FindHeight(Node* root) {
	if (root == nullptr) {
		return -1;
	}
	return max(FindHeight(root->left), FindHeight(root->right)) + 1
}



```

# 10. (Extra) Глубина бинарного дерева
```c++
class Solution {
public:
    int maxDepth(TreeNode* root) {
        if(root == nullptr) {
            return 0;
        }
        int maxLeft = maxDepth(root->left);
        int maxRight = maxDepth(root->right);
        return max(maxLeft, maxRight)+1;
    }
};
```
