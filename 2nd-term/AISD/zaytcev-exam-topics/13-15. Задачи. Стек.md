# 2. Подсчет уникальных элементов стека
камон. заводим хэшмапу.


```c++
#include <iostream>
#include <stack>
#include <unordered_set>

int countUniqueElements(std::stack<int>& stack) {
    std::unordered_set<int> uniqueElements;
    
    while (!stack.empty()) {
        uniqueElements.insert(stack.top());
        stack.pop();
    }
    
    return uniqueElements.size();
}

int main() {
    std::stack<int> myStack;
    myStack.push(5);
    myStack.push(10);
    myStack.push(15);
    myStack.push(10);
    myStack.push(20);
    
    int uniqueCount = countUniqueElements(myStack);
    
    std::cout << "Number of unique elements in the stack: " << uniqueCount << std::endl;
    
    return 0;
}

```



# 3. Составить очередь на стеках
Различие сглаживается с помощью моделирования двумя одноголовочными стеками одной двухголовой очереди.

> - Определим операцию постановки в очередь через добавление в стек.
> - Извлекать из очереди будем выемки из первого стека верхушки и помещения во второй стек.
> - Когда в первом стеке остаётся один элемент, извлекаем его и "переливаем" содержимое второго стека обратно в первый.
> - Две инверсии погасят друг друга, а первый элемент очереди, лежавший на дне стека, в результирующую очередь не попадёт.





# 4. Сравнение стека и очереди
это сравнить поэлементно, и при этом сравнить длину

### При помощи unordered_multiset

```c++
#include <iostream>
#include <stack>
#include <queue>
#include <unordered_set>

bool compareStackAndQueue(std::stack<int>& stack, std::queue<int>& queue) {
    // Create unordered_multiset from stack
    std::unordered_multiset<int> stackSet;
    std::stack<int> tempStack = stack;
    while (!tempStack.empty()) {
        stackSet.insert(tempStack.top());
        tempStack.pop();
    }

    // Create unordered_multiset from queue
    std::unordered_multiset<int> queueSet;
    std::queue<int> tempQueue = queue;
    while (!tempQueue.empty()) {
        queueSet.insert(tempQueue.front());
        tempQueue.pop();
    }

    // Compare the unordered_multisets
    return stackSet == queueSet;
}

int main() {
    std::stack<int> stack;
    std::queue<int> queue;

    // Push elements into stack
    stack.push(20);
    stack.push(10);
    stack.push(30);

    // Push elements into queue
    queue.push(10);
    queue.push(20);
    queue.push(30);

    // Compare the content of stack and queue
    bool sameContent = compareStackAndQueue(stack, queue);

    // Print the result
    if (sameContent) {
        std::cout << "The stack and queue have the same content." << std::endl;
    } else {
        std::cout << "The stack and queue do not have the same content." << std::endl;
    }

    return 0;
}

```


# 5. Сравнение стека и дека

это сравнить поэлементно, и при этом сравнить длину. за хэшмапу плюсик

тут можно использовать решения:  
1) итеративное. цикл по дискретному времени, сравнивающий идентичность элементов.  
2) решение при помощи вектора. пушим в вектор, сравниваем векторы. v1 == v2 - операция, сравнивающая 2 вектора  
3) решение через хеш-таблицу unordered_set.  
эта мапа будет хранить уникальные элементы 20 и 30. она скажет, что таблица одинакова.  
4) unordered_multiset. она сравнит фулл мапу поэлементно.  
и тогда уже скажет, что эти последовательности равны


### Подход простецкий

```c++
1. Итеративный подход к решению
#include <iostream>
#include <stack>
#include <deque>

bool compareStackAndDeque(std::stack<int>& stack, std::deque<int>& deque) {
    if (stack.size() != deque.size()) {
        return false;
    }

    std::stack<int> tempStack = stack;
    std::deque<int> tempDeque = deque;

    while (!tempStack.empty()) {
        if (tempStack.top() != tempDeque.front()) {
            return false;
        }

        tempStack.pop();
        tempDeque.pop_front();
    }

    return true;
}

int main() {
    std::stack<int> stack;
    std::deque<int> deque;

    // Push elements into stack
    stack.push(20);
    stack.push(20);
    stack.push(30);
    stack.push(20);
    stack.push(20);

    // Push elements into deque
    deque.push_back(20);
    deque.push_back(30);
    deque.push_back(20);
    deque.push_back(30);
    deque.push_back(30);

    // Compare the content of stack and deque
    bool sameContent = compareStackAndDeque(stack, deque);

    // Print the result
    if (sameContent) {
        std::cout << "The stack and deque have the same content." << std::endl;
    } else {
        std::cout << "The stack and deque do not have the same content." << std::endl;
    }

    return 0;
}


```



### При помощи unordered_multiset

```c++
2. Подход через unordered_multiset

#include <iostream>
#include <stack>
#include <deque>
#include <unordered_set>

bool compareStackAndDeque(std::stack<int>& stack, std::deque<int>& deque) {
    // Create unordered_multiset from stack
    std::unordered_multiset<int> stackSet;
    std::stack<int> tempStack = stack;
    
    while (!tempStack.empty()) {
        stackSet.insert(tempStack.top());
        tempStack.pop();
    }

    // Create unordered_multiset from deque
    std::unordered_multiset<int> dequeSet(deque.begin(), deque.end());

    // Compare the unordered_multisets
    return stackSet == dequeSet;
}

int main() {
    std::stack<int> stack;
    std::deque<int> deque;

    // Push elements into stack
    stack.push(20);
    stack.push(20);
    stack.push(10);
    stack.push(30);
    stack.push(10);

    // Push elements into deque
    deque.push_back(20);
    deque.push_back(10);
    deque.push_back(20);
    deque.push_back(30);
    deque.push_back(10);

    // Compare the content of stack and deque
    bool sameContent = compareStackAndDeque(stack, deque);

    // Print the result
    if (sameContent) {
        std::cout << "The stack and deque have the same content." << std::endl;
    } else {
        std::cout << "The stack and deque do not have the same content." << std::endl;
    }

    return 0;
}

```
