	size_t size;
	int* array;
public:
	joy(size_t size) : size(size) {
		array = new int[size];
	} // ok

	joy(const joy& other) : size(other.size) {
		array = new int[size];
		for (size_t i = 0; i < size; ++i) {
			array[i] = other.array[i];
		}	
	}
	

	joy& operator=(const joy& other) {
        if (this != &other) { // Проверка на самоприсваивание
            // Освобождение старых ресурсов
	            delete[] array;
            // Выделение новой памяти и копирование данных
	            size = other.size;
	            array = new int[size];
	            for (size_t i = 0; i < size; ++i) {
	                array[i] = other.array[i];
	            }
	        }
		return *this;	
		}

	};
