

В линуксе каждый раз, когда запускаем программу, она оборачивается в процесс.
Для этого процесса выделяется память
```c
#include <stdio.h>
#include <unistd.h>

int main() {
	pid_t pid = fork();
	while (1) {
		printf("[%d] pid returned after fork: %d, parent pid: %d\n",
		getpid(), pid, getppid());
		sleep(10);
	}
}
```

Каждый процесс кем-то создан.
Это превращается в дерево процессов: родитель-ребенок.

```shell
pstree -cp // создает дерево процессов

ctrl f bash

htop // посмотреть статистику процессов

man fork // посмотреть мануал, если есть процессы

```

У каждого процесса есть id.
(тип процесса - pid_t)
Через id можно взаимодействовать с процессом.


После выполнения команды fork создается копия текущего процесса.
Отличается id процесса.
Ребенок выполняет тот же самый код.
Ребенок стартанет с точки fork.

Если pid = 0, мы находимся в ребенке.

```c
getpid() // Вернет собственный id, который мы бы увидели в дереве процесса
getppid() // Вернет id родителя.
```

В ребенке свое адресное пространство, у родителя свое.
Но при этом скопированы.

 
```c
в ребенке безопаснее делать _exit(0); // ничего лишнего не вызовет 

#include <stdio.h>
#include <unistd.h>

int main() {
	int x = 100;
	pid_t pid = fork();
	if (pid == 0) {
		sleep(10);
		printf("[Child] x =  %d\n", x);
		x = 1000;
		_exit(0);
	}
	sleep(10);
	printf("[Parent] x = %d\n", x);
	x = 500;
}

Что будет выведено на экран?

100
100

У каждого процесса своя собственная память, друг от друга процессы не зависят.
Но в самом начале она копируется.
```

- В потоках общая память у всех.
- В процессах своя память. Но изначально одинакова.

#include <stdio.h>
#include <stdlib.h>
#include <sys/wait.h>
#include <unistd.h>

void child_function(){ 
	printf("child pid = %d\n", getpid());
	sleep(500);
	printf("HelloWorld\n");
	_exit(5);
}

int main() {
	pid_t pid = fork();
	if (pid == 0) {
		child_function();
	};
	
	int status; // Это тот самый NULL. Принцип его работы тот же.
	waitpid(pid, &status, 0); // Третий аргумент - опции ожидания. 0 - не хотим доп.
	waitpid(pid1, &status, 0);
	waitpid(pid2, &status, 0); // Статус детей получаем в порядке их завершения
	waitpid(pid3, &status, 0);

	// Проверяем, завершился ли ребенок
	if(WIFEXITED(status)) {
		// Проверка статуса завершения
		printf("Child exit status: %d\n", WEXITSTATUS(status));
	}
	// Проверяем, был ли убит ребенок
	if(WIFSIGANLED(status)) {
		// Номер сигнала, при помощи которого убили
		printf("Child exit status: %d\n", WEXITSTATUS(status));
	}
	
	return 0;
}
