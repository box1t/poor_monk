

# 1. Сравнить стек и дек

это сравнить поэлементно, и при этом сравнить длину


### При помощи unordered_multiset

```c++
2. Подход через unordered_multiset

#include <iostream>
#include <stack>
#include <deque>
#include <unordered_set>

bool compareStackAndDeque(std::stack<int>& stack, std::deque<int>& deque) {
    // Create unordered_multiset from stack
    std::unordered_multiset<int> stackSet;
    std::stack<int> tempStack = stack;
    
    while (!tempStack.empty()) {
        stackSet.insert(tempStack.top());
        tempStack.pop();
    }

    // Create unordered_multiset from deque
    std::unordered_multiset<int> dequeSet(deque.begin(), deque.end());

    // Compare the unordered_multisets
    return stackSet == dequeSet;
}

int main() {
    std::stack<int> stack;
    std::deque<int> deque;

    // Push elements into stack
    stack.push(20);
    stack.push(20);
    stack.push(10);
    stack.push(30);
    stack.push(10);

    // Push elements into deque
    deque.push_back(20);
    deque.push_back(10);
    deque.push_back(20);
    deque.push_back(30);
    deque.push_back(10);

    // Compare the content of stack and deque
    bool sameContent = compareStackAndDeque(stack, deque);

    // Print the result
    if (sameContent) {
        std::cout << "The stack and deque have the same content." << std::endl;
    } else {
        std::cout << "The stack and deque do not have the same content." << std::endl;
    }

    return 0;
}

```



# 2. Слияние деков

это 2 строчки.
из одного вытащил, в другой вставил



### Решение 1. Без дополнительного дека

Это решение более эффективно по времени и по памяти, чем при создании 3 дека, в который могли бы добавлять элементы из 1 и 2.

>The time and memory complexity of merging two deques using the provided algorithm can be analyzed as follows:

>Time Complexity:
- Inserting an element at the end of a deque (`push_back`) takes constant time on average.
- In the `mergeDeques` function, we iterate over each element in `deque2` and insert it into `deque1` using `push_back`. This operation is performed `n` times, where `n` is the size of `deque2`.
- Therefore, the time complexity of the merging process is O(n), where n is the size of `deque2`.

> Memory Complexity:
- The memory complexity of merging two deques depends on whether we create a new deque or merge the elements into one deque.
- In the initial version of the code that creates a new deque, the memory complexity is O(m + n), where m and n are the sizes of `deque1` and `deque2`, respectively. This is because we create a new deque to store all the elements from both `deque1` and `deque2`.
- In the updated version of the code that merges the elements into one deque, the memory complexity is O(1). This is because we don't create any additional data structures or allocate extra memory. We simply modify `deque1` by inserting elements from `deque2` directly into it.

In summary:
- Time complexity: O(n), where n is the size of `deque2`.
- Memory complexity:
  - Initial version: O(m + n), where m and n are the sizes of `deque1` and `deque2`, respectively.
  - Updated version: O(1), as we modify `deque1` without creating any additional data structures.

```c++
#include <iostream>
#include <deque>

using namespace std;

void mergeDeques(deque<int>& deque1, const deque<int>& deque2) {
    // Insert elements from deque2 into deque1
    for (const auto& element : deque2) {
        deque1.push_back(element);
    }
}

int main() {
    deque<int> deque1 = {1, 2, 3};
    deque<int> deque2 = {4, 5, 6};

    mergeDeques(deque1, deque2);

    cout << "Merged Deque: ";
    for (const auto& element : deque1) {
        cout << element << " ";
    }
    cout << endl;

    return 0;
}

```

### Решение 2. Нубское, создаём 3 дек

```c++
#include <iostream>
#include <deque>

using namespace std;

deque<int> mergeDeques(const deque<int>& deque1, const deque<int>& deque2) {
    deque<int> mergedDeque;

    // Insert elements from deque1
    for (const auto& element : deque1) {
        mergedDeque.push_back(element);
    }

    // Insert elements from deque2
    for (const auto& element : deque2) {
        mergedDeque.push_back(element);
    }

    return mergedDeque;
}

int main() {
    deque<int> deque1 = {1, 2, 3};
    deque<int> deque2 = {4, 5, 6};

    deque<int> mergedDeque = mergeDeques(deque1, deque2);

    cout << "Merged Deque: ";
    for (const auto& element : mergedDeque) {
        cout << element << " ";
    }
    cout << endl;

    return 0;
}
```

# 3. Удаление элемента из дека

```c++
#include <iostream>
#include <deque>

using namespace std;

void deleteElement(deque<int>& deque1, int element) {
    for (auto it = deque1.begin(); it != deque1.end(); ++it) {
        if (*it == element) {
            deque1.erase(it);
            break;
        }
    }
}

int main() {
    deque<int> deque1 = {1, 2, 3, 4, 5};

    int elementToDelete = 3;

    deleteElement(deque1, elementToDelete);

    cout << "Updated Deque: ";
    for (const auto& element : deque1) {
        cout << element << " ";
    }
    cout << endl;

    return 0;
}
```

# 4. Подсчет различных элементов дека


```c++
#include <iostream>
#include <deque>
#include <unordered_set>

int countUniqueElements(std::deque<int>& deque) {
    std::unordered_set<int> uniqueElements;
    
    while (!deque.empty()) {
        uniqueElements.insert(deque.front());
        deque.pop_front();
    }
    
    return uniqueElements.size();
}

int main() {
    std::deque<int> myDeque;
    myDeque.push_front(5);
    myDeque.push_front(10);
    myDeque.push_front(15);
    myDeque.push_front(10);
    myDeque.push_front(20);
    
    int uniqueCount = countUniqueElements(myDeque);
    
    std::cout << "Number of unique elements in the deque: " << uniqueCount << std::endl;
    
    return 0;
}

```

```c++
#include <iostream>

int countDifferentElements(const deque<int>& deque1) {
    unordered_set<int> uniqueElements;

    for (const auto& element : deque1) {
        uniqueElements.insert(element);
    }

    return uniqueElements.size();
}

int main() {
    deque<int> deque1 = {1, 2, 3, 2, 4, 1, 5, 3};

    int numDifferentElements = countDifferentElements(deque1);

    cout << "Number of different elements: " << numDifferentElements << endl;
```
