```c++

#include <iostream>
#include <queue>

template <typename T>
void Push(std::queue<T>& q, const T& t) {
    std::queue<T> q1;

    // Copy elements from q to q1
    while (!q.empty()) {
        q1.push(q.front());
        q.pop();
    }

    // Copy elements from q1 back to q
    while (!q1.empty()) {
        q.push(q1.front());
        q1.pop();
    }

    q.push(t);  // Add t to q
}

int main() {
    std::queue<int> myQueue;
    Push(myQueue, 42);  // Pushing value 42 into the queue

    while (!myQueue.empty()) {
        std::cout << myQueue.front() << " ";
        myQueue.pop();
    }
    std::cout << std::endl;

    return 0;
}

```

***Функция ПЕРВЫЙ.*** 
В соответствии с выбранным отображением первый элемент всегда в начале файла.
Несмотря на наличие **var**, очередь не модифицируется. Элемент берётся из буфера в памяти q |.

```c++

#include <iostream>
#include <queue>

template <typename T>
T Top(std::queue<T>& q) {
    T topElement;

    if (!q.empty()) {
        topElement = q.front();
    }

    return topElement;
}

int main() {
    std::queue<int> myQueue;
    myQueue.push(10);
    myQueue.push(20);
    myQueue.push(30);

    int topElement = Top(myQueue);

    if (!myQueue.empty()) {
        std::cout << "Top element: " << topElement << std::endl;
    } else {
        std::cout << "Queue is empty." << std::endl;
    }

    return 0;
}

```

Функция ***ИЗ_ОЧЕРЕДИ*** обеспечивает исключение компоненты из файла, хранящего очередь.
Альтернативное решение - сделать reset(q) и пропустить get(q) не годится, т.к.

```c++

#include <iostream>
#include <queue>

template <typename T>
void Pop(std::queue<T>& q) {
    std::queue<T> q1;

    if (!q.empty()) {
        q.pop();  // Skip the first element if it exists
    }

    // Copy the remaining elements of q (without the first element) to q1
    while (!q.empty()) {
        q1.push(q.front());
        q.pop();
    }

    // Copy the modified queue q1 (without the first element) back to q
    while (!q1.empty()) {
        q.push(q1.front());
        q1.pop();
    }
}

int main() {
    std::queue<int> myQueue;
    myQueue.push(10);
    myQueue.push(20);
    myQueue.push(30);

    Pop(myQueue);  // Pop the first element from the queue

    while (!myQueue.empty()) {
        std::cout << myQueue.front() << " ";
        myQueue.pop();
    }
    std::cout << std::endl;

    return 0;
}


```
