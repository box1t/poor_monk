### 1 Функция Destroy()

==чтение очереди разрушающее - это удаление. это нормальное чтение.==

Для уничтожения очереди поочерёдно удаляем все её элементы, сохраняя в локальной перемеенной pi (на мгновение!) перед удалением каждого элемента содержащуюся в нём ссылку на следующий элемент.

```c

void Destroy(queue* q)
{
	while (!Empty(q))
	{
		struct Item* pi = q->first;
		q->first = q->first->next;
		free(pi);
	}
free(q->first);
q->first = q->last = 0;
q->size = 0;
}

```

### 2 Рекурсивная программа Destroy

Существуют итеративная и рекурсивная процедуры Destroy.

Рекурсивная совмещает *обход* очереди и посещение всех её элементов с одновременным удалением пройденных компонент.


```c

void Destroy(queue* q)
{
	q->last->next = 0;
	struct Item* pi = q->first;
	q->first = q->first->next; // актуальное начало остатка очереди
	free(pi); 
	if(q->first)
		Destroy(q);
	q->last = 0;
	q->size = 0;
}

```

# 3. Сортировка очереди

Сортировка вставкой

```c++
#include <iostream>
#include <queue>

using namespace std;

// Function to insert an element at the correct position in the queue
void insertSorted(queue<int>& q, int element) {
    // If the queue is empty or the new element is greater than the last element in the queue, simply push it to the back
    if (q.empty() || element > q.back()) {
        q.push(element);
        return;
    }

    // Otherwise, find the correct position to insert the element
    // by temporarily storing the elements in another queue
    queue<int> tempQ;

    while (!q.empty() && q.front() < element) {
        tempQ.push(q.front());
        q.pop();
    }

    // Insert the new element
    q.push(element);

    // Move the remaining elements back to the original queue
    while (!tempQ.empty()) {
        q.push(tempQ.front());
        tempQ.pop();
    }
}

// Function to print the contents of the queue
void printQueue(queue<int>& q) {
    while (!q.empty()) {
        cout << q.front() << " ";
        q.pop();
    }
    cout << endl;
}

int main() {
    queue<int> myQueue;

    // Insert elements into the queue
    insertSorted(myQueue, 5);
    insertSorted(myQueue, 3);
    insertSorted(myQueue, 8);
    insertSorted(myQueue, 1);
    insertSorted(myQueue, 2);

    // Print the sorted queue
    printQueue(myQueue);

    return 0;
}

```

# 5. Сравнение стека и очереди


```c++
#include <iostream>
#include <stack>
#include <queue>
#include <unordered_set>

bool compareStackAndQueue(std::stack<int>& stack, std::queue<int>& queue) {
    // Create unordered_multiset from stack
    std::unordered_multiset<int> stackSet;
    std::stack<int> tempStack = stack;
    while (!tempStack.empty()) {
        stackSet.insert(tempStack.top());
        tempStack.pop();
    }

    // Create unordered_multiset from queue
    std::unordered_multiset<int> queueSet;
    std::queue<int> tempQueue = queue;
    while (!tempQueue.empty()) {
        queueSet.insert(tempQueue.front());
        tempQueue.pop();
    }

    // Compare the unordered_multisets
    return stackSet == queueSet;
}

int main() {
    std::stack<int> stack;
    std::queue<int> queue;

    // Push elements into stack
    stack.push(20);
    stack.push(10);
    stack.push(30);

    // Push elements into queue
    queue.push(10);
    queue.push(20);
    queue.push(30);

    // Compare the content of stack and queue
    bool sameContent = compareStackAndQueue(stack, queue);

    // Print the result
    if (sameContent) {
        std::cout << "The stack and queue have the same content." << std::endl;
    } else {
        std::cout << "The stack and queue do not have the same content." << std::endl;
    }

    return 0;
}

```
