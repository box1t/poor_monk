

Алгоритм Рутисхаузера - один из старейших примитивных алгоритмов выражений.
Тогда ещё не было языков программирования, только МТ.
В этом его особенность.
Не было рекурсии и деревьев.

Раз этот ранний алгоритм несовершенен, там должны быть расставлены полные скобочные структуры.

***Этапы алгоритма Рутисхаузера:***
1. Расставить уровни по литерам из входного потока.
2. Отыскать элементы строки с максимальным значением уровня
3. Выделить тройку - два операнда с макс. значением уровня и операцию между ними.
4. Результат операции обозначить вспомогательной переменной
5. Из строки удалить тройку с её скобками, поставить на её место нашу переменную.
6. Выполнять пункты 2-5 до получения переменной-результата.

- Открывающая скобка добавляет уровни, операнды тоже. 
- Операции и закрывающие скобки - убавляют. 

В результате ныряния в ямку рельеф мельчает (уровень падает), а выныривая из неё, уровень повышаем.

***Эффективность алгоритма:***
Поиск самой глубокой ямку стоит n/2, а уничтожаем за O(1), затем сдвигаем выражение за n/2. 
У нас есть линейное время на расстановку уровней и линейное время на отработку каждой ямки.

### Реализация алгоритма на Си

Тест: ((((А + B) * C ) / D ) - E )
0 1 2 3 4 5 4 5 4 3 4 3 2 3 2 1 2 1 0

```c
// Полная скобочная структура, только числа и из операций: + - * /

#include <stdio.h>
#include <string.h>
#include <stdlib.h>
int isOp(char c) { //  Проверка на операцию
    return (c == '+') || (c == '-') || (c == '*') || (c == '/');
}

int isNum(char c) { // Проверка на цифру
    return c >= '0' && c <= '9';
}

double doOp(char op, double a, double b) { // Выполнить операцию
    switch(op) {
        case '+':
            return a + b;
        case '-':
            return a - b;
        case '*':
            return a * b;
        case '/':
            return a / b;
    }
}

int main() {
    int N = 100;
    char * s = (char*)malloc(sizeof(char)*N); // Задаём строку через malloc, чтобы можно было менять строку
    scanf("%s", s);
    int M = strlen(s);// Размер строки
    int d[M];// Массив уровней
    double nums[M];// Массив с числами. Тип double, чтобы можно было корректно делить
    int i = 0, j = 0, depth = 0, numc = 0;// i, j - два итератора(Ниже объясню зачем), depth - уровень, numc - количество цифр до тек. позиции
    while(i < M)
    {
        if(s[i] == '(')// Если текущий символ - открывающаяся скобка или число, то увеличиваем уровень
        // В противном случае - понижаем
        {
            depth++;
        }
        else if((s[i] == ')') || isOp(s[i]))
        {
            depth--;
        }
        else// Встретили число - как обычно собираем его посимвольно и добавляем в массив чисел
        {
            depth++;
            double buff = s[i] - '0';
            while(isNum(s[i+1]))
            {
                buff = buff*10 + (s[i+1] - '0');
                i++;// Накапливаем i, чтобы после сборки числа он указывал на конец числа
            }
            nums[numc] = buff;
            numc++;
        }
        s[j] = s[i];// По сути j всегда увеличивается на 1, а i прыгает через числа.
        // В итоге у нас числа "схлопываются" последнюю свою цифру 12345 -> 5
        d[j] = depth;
        i++;// Увеличиваем i и j синхронно
        j++;
    }
    s[j] = '\0';// Устанавливаем терминатор строки, т.к. в результате "схлопываний" чисел, дальше идет ненужный мусор
    M = strlen(s);// Обновляем размер строки
    for(int k = 0; k < M; k++)
    {
        if(isNum(s[k]))// Заменяем последнюю цифру "схлопнутого" числа условным знаком "n"
        {
            s[k] = 'n';
        }
    }
    while(M > 1)// Пока длина строки больше 1 => можно проводить операции
    {
        printf("%s\n", s);
        int k = -1, nc = -1;
        for(i = 1; i < M-1; i++)// Ищем максимальный пик и индекс числа перед оператором с максимальным пиком
        {
            if((d[i-1] > d[i]) && (d[i+1] > d[i])) {
                if((k == -1) || (d[i] > d[k])) {
                    k = i;
                    nc++;
                }
            }
        }
        for(int z = 0; z < k; z++) {
            putchar(' ');
        }
        printf("^\n");
        double res = doOp(s[k], nums[nc], nums[nc+1]);// Считаем результат операции и записываем результат в массив чисел
        nums[nc] = res;
        int x;
        for(x = nc+1; x < numc-1; x++)// Сдвигаем ост. часть чисел на 1 влево
        {
            nums[x] = nums[x+1];
        }

        numc--;

        s[k-2] = 'n';// Аналогичным образом обрабатываем строку выражения и массив уровней (n+n) -> n

        d[k-2] = d[k];
        for(x = k-1; x < M-4; x++)
        {
            s[x] = s[x+4];
            d[x] = d[x+4];
        }
        s[x] = '\0';// Уст. терминаторы, чтобы отсечь возможный мусор
        d[x] = '\0';
        M = strlen(s);// Обновляем длину строки
    }
    printf("%s\n", s);// Выводим последний вариант строки, т.е. n и результат всего выражения
    printf("%lf\n", nums[0]);// Т.к. "схлопывание" массива чисел не работает, когда их всего 2, просто выводим самое первое.
    return 0;
}
```
