using std:: cin;

{
    int n, d=1;
    cin>>n;
    bool fl = true;
    for (int x=2; x<1000000; x++)
    {
        fl=true;
        for (int i=2; i<sqrt(x)+1; i++)
        {
            if (x%i==0) fl=false;

Задание №15:
Вывести на экран вначале все единицы из двоичного разложения числа, введенного с клавиатуры, потом все нули.

#include <iostream>
using std ::cout;
using std ::endl;
using std ::cin;
int main()
{
    int n, c0=0, c1=0, a=1;
    cin>>n;
    while (2*a<=n) a=a*2;
    while (a>0)
    {
        if (n>=a)
        {
            c1++;
            n=n-a;
        }
        else
        {
            c0++;
        }
        a=a/2;
    }
    for (int i=0; i<c1;i++) cout<<1;
    for (int d=0; d<c0;d++) cout<<0;
    return 0;
}

Задание №17:
Дано N цилиндров, все они разного размера и из разных материалов. Нужно определить, который из них окажется самым тяжёлым.

using std ::cout;
using std ::endl;
using std ::cin;
int main()
{
    int maxi, n;
    float R, h, p, maxm=0;
    cin>>n;
    for (int i=0; i<n; i++)
    {
        cin >> R >> h >> p;
        if (R*R*h*p>maxm)
        {
            maxm=R*R*h*p;
            maxi=i;
        }
    }
    cout<<maxi;
    return 0;
}

Задание №18:
Определите, сколько раз в последовательности целых чисел встречается ее максимальный элемент. Целое число n - количество элементов (не более 1000000). Набор из n целых чисел.

#include <iostream>
using std ::cout;
using std ::endl;
using std ::cin;
int main()
{
    int n, k=0, max=-99999, d;
    cin>>n;
    for (int i=0; i<n; i++)
    {
        cin>>d;
        if (d>max)
        {
            max=d;
            k=1;
        }
        else if (d==max)
        {
            k++;
        }
    }
    cout<<k<<endl;
    return 0;
}
